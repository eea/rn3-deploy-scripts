# Quick Guide 

# GET THE CODE 
cd /$your_prefered_directory
git clone https://github.com/eea/rn3-deploy-scripts.git .

#git checkout sotav-NKP

cd NKP/eea-deployer/


# Naming conventions 
the value files to be used need to have the names like this : config-$ENV-$CLUSTER.yaml
values of CLUSTER: "nkp", "ran", "ran2", "pre"
values of ENV: "dev", "test", "staging", "sandbox", "transport", "prod"
ex: config-staging-nkp.yaml
    config-transport-nkp.yaml
    config-dev-ran.yaml
    config-sandbox-nkp.yaml
    config-test-ran.yaml
    config-prod-prem.yaml




# first thing to do !!!!! after git clone

WARNING : every value file needs to have the correct kubeconfig file to the cluster it targets 
edit the config file and put 'kc: $YOUR_DIRECTORY_PATH_OF_THE_KUBECONFIG' on the main block of the file (at the start....)
ex: kc: /home/$USER/.kube/staging/config

TIP: in case you use a single config file make sure before you run the script to have selected 
the correct context you need from the config using : kubectl config get-contexts and kubectl config use-context $YOUR_CONTEXT
ex :    #dev
        kubectl config use-context sotirios.tavoulas-kavoulas-172.22.145.232
        #prod
        kubectl config use-context sotirios.tavoulas-kavoulas-172.22.145.202

edit the config file and put 'ns: $YOUR_NAMESPACE' on the main block of the file (at the start....)
ex: ns: reportnet

Usage : ./eea-deployer $ACTION -f ./$config
ex: ./eea-deployer install -f ./config-staging-nkp.yaml
    ./eea-deployer install -f ./config-transport-nkp.yaml
    ./eea-deployer uninstall -f ./config-staging-nkp.yaml
    ./eea-deployer uninstall -f ./config-transport-nkp.yaml


# set an app to  get deployed to another namespace insted of the global one 
edit the config file go to the block of the application and set 'namespace: $YOUR_NAMESPACE'
ex: namespace: reportnet

# set an app to not get deployed at all
edit the config file and put 'deploy_type = "-"'

# set a deployment to run any command before the deployment 
edit the config file go to the block of the application and set 'init_shell: >- $commands; '

# set an app to run any command after the deployment 
edit the config file go to the block of the application and set 'script_shell: >- $commands; '

# set an app to get deployed with a custom command line 
edit the config file go to the block of the application and set 'custom_deployment: >- $commands; '
